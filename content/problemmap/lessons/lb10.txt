```python
def beautiful_pairs(nums1: list[int], nums2: list[int]) -> list[int]:
    first: dict[tuple[int, int], int] = {}
    best = (10**18, 0, 0)                  # (distance, i, j)
    pts = []
    for i, p in enumerate(zip(nums1, nums2)):
        if p in first:                     # equal points: distance 0
            best = min(best, (0, first[p], i))
        else:
            first[p] = i
            pts.append((p[0], p[1], i))
    pts.sort()
    def solve(lo: int, hi: int) -> list:   # returns the points sorted by y
        nonlocal best
        if hi - lo <= 1:
            return pts[lo:hi]
        mid = (lo + hi) // 2
        mx = pts[mid][0]
        by_y = sorted(solve(lo, mid) + solve(mid, hi), key=lambda p: p[1])
        strip: list = []
        for p in by_y:
            if abs(p[0] - mx) <= best[0]:  # inside the thin strip
                for q in reversed(strip):
                    if p[1] - q[1] > best[0]:
                        break
                    d = abs(p[0] - q[0]) + abs(p[1] - q[1])
                    best = min(best, (d, min(p[2], q[2]), max(p[2], q[2])))
                strip.append(p)
        return by_y
    solve(0, len(pts))
    return [best[1], best[2]]
```

```python
import math, random

def outer_trees2(trees: list[list[int]]) -> list[float]:
    pts = [tuple(t) for t in trees]
    random.shuffle(pts)                    # random order keeps the expected time O(n)
    def inside(c, p):
        return math.dist(c[:2], p) <= c[2] + 1e-9
    def two(a, b):                         # circle with a-b as its diameter
        return ((a[0] + b[0]) / 2, (a[1] + b[1]) / 2, math.dist(a, b) / 2)
    def three(a, b, c):                    # circle through three points
        bx, by, cx, cy = b[0] - a[0], b[1] - a[1], c[0] - a[0], c[1] - a[1]
        d = 2 * (bx * cy - by * cx)
        ux = (cy * (bx * bx + by * by) - by * (cx * cx + cy * cy)) / d
        uy = (bx * (cx * cx + cy * cy) - cx * (bx * bx + by * by)) / d
        return (a[0] + ux, a[1] + uy, math.hypot(ux, uy))
    c = (pts[0][0], pts[0][1], 0.0)
    for i, p in enumerate(pts):
        if inside(c, p):
            continue
        c = (p[0], p[1], 0.0)              # p must lie on the edge
        for j in range(i):
            if inside(c, pts[j]):
                continue
            c = two(p, pts[j])             # p and pts[j] are both on the edge
            for k in range(j):
                if not inside(c, pts[k]):
                    c = three(p, pts[j], pts[k])
    return list(c)
```

```python
import cmath

def multiply(poly1: list[int], poly2: list[int]) -> list[int]:
    need = len(poly1) + len(poly2) - 1
    size = 1
    while size < need:
        size *= 2
    def fft(a: list[complex], sign: int) -> list[complex]:
        n = len(a)
        if n == 1:
            return a
        even, odd = fft(a[0::2], sign), fft(a[1::2], sign)
        out = [0j] * n
        for k in range(n // 2):
            w = cmath.exp(sign * 2j * cmath.pi * k / n) * odd[k]
            out[k], out[k + n // 2] = even[k] + w, even[k] - w
        return out
    fa = fft([complex(x) for x in poly1] + [0j] * (size - len(poly1)), 1)
    fb = fft([complex(x) for x in poly2] + [0j] * (size - len(poly2)), 1)
    prod = fft([x * y for x, y in zip(fa, fb)], -1)   # inverse: needs / size
    return [round(prod[i].real / size) for i in range(need)]
```

@@ diameter
what: The longest path in a tree bends at one node and goes down two branches. So for every node, find the depth of its two deepest child branches. The best path through the node is their sum. Return only the deepest branch to the parent, and keep the best sum seen in a variable.

```python
class Node:
    def __init__(self, val=None, children=None):
        self.val = val
        self.children = children or []

def diameter(root: "Node") -> int:
    best = 0
    def depth(node: "Node") -> int:
        nonlocal best
        top1 = top2 = 0                    # two deepest child branches
        for ch in node.children:
            d = depth(ch) + 1
            if d > top1:
                top1, top2 = d, top1
            elif d > top2:
                top2 = d
        best = max(best, top1 + top2)      # path bending at this node
        return top1
    if root:
        depth(root)
    return best
```

```python
from collections import deque, defaultdict

def find_ladders(begin_word: str, end_word: str, word_list: list[str]) -> list[list[str]]:
    words = set(word_list)
    if end_word not in words:
        return []
    dist = {begin_word: 0}
    parents: dict[str, list[str]] = defaultdict(list)
    q = deque([begin_word])
    while q:
        w = q.popleft()
        if w == end_word:
            break
        for i in range(len(w)):
            for c in "abcdefghijklmnopqrstuvwxyz":
                nxt = w[:i] + c + w[i + 1:]
                if nxt not in words:
                    continue
                if nxt not in dist:
                    dist[nxt] = dist[w] + 1
                    q.append(nxt)
                if dist[nxt] == dist[w] + 1:     # w is on a shortest path to nxt
                    parents[nxt].append(w)
    paths: list[list[str]] = []
    def back(w: str, tail: list[str]) -> None:
        if w == begin_word:
            paths.append([w] + tail[::-1])
            return
        for p in parents[w]:
            back(p, tail + [w])
    if end_word in dist:
        back(end_word, [])
    return paths
```

```python
from collections import deque

def cat_mouse_game(graph: list[list[int]]) -> int:
    n = len(graph)
    MOUSE, CAT = 0, 1                      # whose turn it is
    res = {}                               # (mouse, cat, turn) -> 1 mouse wins, 2 cat wins
    degree = {}                            # moves not yet known to be bad
    q = deque()
    for m in range(n):
        for c in range(1, n):
            degree[m, c, MOUSE] = len(graph[m])
            degree[m, c, CAT] = len([x for x in graph[c] if x != 0])
    for c in range(1, n):
        for t in (MOUSE, CAT):
            res[0, c, t] = 1
            q.append((0, c, t, 1))         # mouse reached the hole
            res[c, c, t] = 2
            q.append((c, c, t, 2))         # cat caught the mouse
    while q:
        m, c, t, r = q.popleft()
        prev_t = 1 - t
        for p in (graph[m] if prev_t == MOUSE else graph[c]):
            pm, pc = (p, c) if prev_t == MOUSE else (m, p)
            if pc == 0 or (pm, pc, prev_t) in res:
                continue
            if (prev_t == MOUSE and r == 1) or (prev_t == CAT and r == 2):
                res[pm, pc, prev_t] = r    # the mover can pick this win
                q.append((pm, pc, prev_t, r))
            else:
                degree[pm, pc, prev_t] -= 1
                if degree[pm, pc, prev_t] == 0:   # every move is bad
                    res[pm, pc, prev_t] = r
                    q.append((pm, pc, prev_t, r))
    return res.get((1, 2, MOUSE), 0)       # 0 means draw
```

@@ mono-dp
what: Some DPs need the best earlier state, or the cheapest partner to merge with, and a plain loop would scan back each time. A monotonic stack keeps only the candidates that can still win, in sorted order. Each item is pushed and popped once, so every lookup is amortized O(1).
signals: build a tree or merge items using the max of each part ; the next greater or smaller item is the useful partner ; minimize a sum of products of neighbours ; the DP looks left for the nearest bigger or smaller value ; n up to 40 or more, where interval DP is too slow
code:
```python
def mct_from_leaf_values(arr: list[int]) -> int:
    stack = [float("inf")]
    total = 0
    for x in arr:
        while stack[-1] <= x:
            mid = stack.pop()            # smallest leaf is merged first
            total += mid * min(stack[-1], x)
        stack.append(x)
    while len(stack) > 2:                # leftovers merge with their left neighbour
        total += stack.pop() * stack[-1]
    return total
```

@@ dpstate2d
what: When a choice can be made only a limited number of times, add the count to the DP state. The state becomes the last choice together with how many uses are left or done. Each state then has a simple rule from the previous one, and the table is small because the counter is small.
signals: at most k transactions, groups, or switches ; the answer depends on how many times you already used an action ; k is small, such as 2, or at most about 100 ; one state per item and per count ; each use of the action changes the state by one
code:
```python
def max_profit(prices: list[int]) -> int:
    K = 2
    buy = [float("-inf")] * (K + 1)    # best cash while holding a stock after j buys
    sell = [0] * (K + 1)               # best cash with no stock after j sells
    for p in prices:
        for j in range(1, K + 1):
            buy[j] = max(buy[j], sell[j - 1] - p)
            sell[j] = max(sell[j], buy[j] + p)
    return sell[K]
```

@@ strdp
what: Many string DPs ask whether piece s[i:i+L] equals the next piece s[i+L:i+2L]. Comparing characters costs O(L) every time. A table of longest common prefix lengths, or a hash, answers each comparison in O(1). Then the DP only decides which cuts are allowed and takes the best result.
signals: delete or split a prefix when it equals the next part ; the rule is that two pieces of the string are equal ; maximum number of operations on a string ; a DP over positions where each step tries many piece lengths ; n up to about 4000, so O(n²) states are fine but O(n³) is not
code:
```python
def delete_string(s: str) -> int:
    n = len(s)
    lcp = [[0] * (n + 1) for _ in range(n + 1)]   # lcp[i][j]: common prefix of s[i:] and s[j:]
    for i in range(n - 1, -1, -1):
        for j in range(n - 1, i, -1):
            if s[i] == s[j]:
                lcp[i][j] = lcp[i + 1][j + 1] + 1
    dp = [1] * (n + 1)                 # dp[i]: most operations for s[i:]
    for i in range(n - 1, -1, -1):
        for L in range(1, (n - i) // 2 + 1):
            if lcp[i][i + L] >= L:     # s[i:i+L] equals the next L characters
                dp[i] = max(dp[i], 1 + dp[i + L])
    return dp[0]
```

@@ convex-search
what: A convex cost function goes down and then up, with one lowest point. Compare the cost at m with the cost at m + 1. If it still goes down, the minimum is to the right. If not, it is at m or to the left. Each comparison halves the range, so you need about log(range) cost checks.
signals: choose one value x to minimize a total cost ; the cost is a sum of weighted distances ; the cost goes down and then up as x moves ; x can be any integer in a range ; checking one x is easy but trying every x is too slow
code:
```python
def min_cost(nums: list[int], cost: list[int]) -> int:
    def total(x: int) -> int:
        return sum(abs(v - x) * c for v, c in zip(nums, cost))
    lo, hi = min(nums), max(nums)
    while lo < hi:
        mid = (lo + hi) // 2
        if total(mid) <= total(mid + 1):   # not going down: the minimum is at mid or left
            hi = mid
        else:
            lo = mid + 1
    return total(lo)
```

@@ segtree-descend
what: Store the maximum of each range in a segment tree node. To find the first position with a value that fits, start at the root and go left whenever the left half has a big enough maximum. Otherwise go right. This finds the leftmost fit in O(log n), instead of scanning from the left.
signals: find the first or leftmost position where value >= x ; each item uses up a slot, so that slot changes later ; n and the number of queries up to 10^5 ; place items in the first available box ; a plain scan per query would be O(n²) in total

```python
def num_of_unplaced_fruits(fruits: list[int], baskets: list[int]) -> int:
    n = len(baskets)
    size = 1
    while size < n:
        size *= 2
    tree = [0] * (2 * size)            # tree[node] = largest basket in that range
    for i, b in enumerate(baskets):
        tree[size + i] = b
    for i in range(size - 1, 0, -1):
        tree[i] = max(tree[2 * i], tree[2 * i + 1])
    unplaced = 0
    for f in fruits:
        if tree[1] < f:                # no basket anywhere is big enough
            unplaced += 1
            continue
        node = 1
        while node < size:             # go left whenever the left half can fit
            node = 2 * node if tree[2 * node] >= f else 2 * node + 1
        tree[node] = 0                 # the basket is used
        node //= 2
        while node:
            tree[node] = max(tree[2 * node], tree[2 * node + 1])
            node //= 2
    return unplaced
```

@@ fenwick-dp
what: Some DPs need the best value among earlier items whose key is not larger than the current key. A plain loop over earlier items is O(n) per item. A Fenwick tree that stores prefix maximums gives the answer in O(log n). Update it after each item, and the DP runs in O(n log n).
signals: best subsequence where each step must satisfy a rule on a key such as nums[i] - i ; longest or heaviest increasing subsequence with n up to 10^5 ; dp[i] needs max of dp[j] over j < i with key[j] <= key[i] ; the rule can be rewritten as one number per item ; an O(n²) DP is too slow
code:
```python
def max_balanced_subsequence_sum(nums: list[int]) -> int:
    keys = sorted({v - i for i, v in enumerate(nums)})
    rank = {k: r + 1 for r, k in enumerate(keys)}
    tree = [0] * (len(keys) + 1)       # Fenwick tree for prefix maximum; 0 means start fresh
    def query(i: int) -> int:
        best = 0
        while i > 0:
            best = max(best, tree[i])
            i -= i & -i
        return best
    def update(i: int, val: int) -> None:
        while i < len(tree):
            tree[i] = max(tree[i], val)
            i += i & -i
    ans = float("-inf")
    for i, v in enumerate(nums):
        r = rank[v - i]
        cur = v + query(r)             # best chain ending earlier with key <= this key
        ans = max(ans, cur)
        update(r, cur)
    return ans
```

```python
import sqlite3

def biggest_window(rows: list[tuple[int, str]]) -> list[tuple[int, int]]:
    db = sqlite3.connect(":memory:")
    db.execute("CREATE TABLE UserVisits (user_id INT, visit_date TEXT)")
    db.executemany("INSERT INTO UserVisits VALUES (?, ?)", rows)
    query = """
    SELECT user_id, MAX(gap) AS biggest_window
    FROM (
        SELECT user_id,
               CAST(julianday(COALESCE(
                   LEAD(visit_date) OVER (PARTITION BY user_id ORDER BY visit_date),
                   '2021-01-01')) - julianday(visit_date) AS INT) AS gap
        FROM UserVisits
    )
    GROUP BY user_id ORDER BY user_id"""
    return db.execute(query).fetchall()
```

@@ diff-candidates
what: When every pair of items can be fixed by changing 0, 1 or 2 values, the cost depends on the target value in ranges. For each pair, add its cost to those ranges in a difference array, with only two writes per range. One prefix sum over the array then gives the cost of every possible target. Take the minimum.
signals: make all pairs sum to the same target ; cost is 0, 1 or 2 changes depending on the target ; the target ranges over a small interval, such as 2 to 2 * limit ; each pair adds a cost to a range of targets ; limit up to 10^5, so trying each target with each pair is too slow
code:
```python
def min_moves(nums: list[int], limit: int) -> int:
    diff = [0] * (2 * limit + 2)       # diff[t]: change in cost when the target moves to t
    n = len(nums)
    for i in range(n // 2):
        a, b = nums[i], nums[n - 1 - i]
        lo, hi = min(a, b), max(a, b)
        diff[2] += 2                   # start: every target costs 2
        diff[lo + 1] -= 1              # from lo+1 on, one change is enough
        diff[a + b] -= 1               # exactly a+b costs 0
        diff[a + b + 1] += 1
        diff[hi + limit + 1] += 1      # past hi+limit, two changes again
    best = float("inf")
    cost = 0
    for t in range(2, 2 * limit + 1):
        cost += diff[t]
        best = min(best, cost)
    return best
```

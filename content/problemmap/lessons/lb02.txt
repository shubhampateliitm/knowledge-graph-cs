        if nums[mid] < target:

def last_stone_weight(stones: list[int]) -> int:
    heap = [-s for s in stones]        # negate: the smallest negative is the largest stone

@@ subsets
what: For each item you make one choice: put it in the current subset or leave it out. Following both choices for all n items gives a tree with 2^n leaves, and each leaf is one subset. def subsets(nums: list[int]) -> list[list[int]]:
    res: list[list[int]] = []
    cur: list[int] = []
    def go(i: int) -> None:
        if i == len(nums):
            res.append(cur[:])         # copy: cur keeps changing
            return
        cur.append(nums[i])            # choice 1: include nums[i]
        go(i + 1)
        cur.pop()                      # undo the choice
        go(i + 1)                      # choice 2: exclude nums[i]
    go(0)
    return res
```

    i, j, carry = len(a) - 1, len(b) - 1, 0
    out: list[str] = []
    while i >= 0 or j >= 0 or carry:
        total = carry
        if i >= 0:
            total += int(a[i]); i -= 1
        if j >= 0:
            total += int(b[j]); j -= 1
        out.append(str(total % 2))     # digit to write
        carry = total // 2             # digit to carry
    return "".join(reversed(out))      # digits were produced right to left

def check_straight_line(coordinates: list[list[int]]) -> bool:
    x0, y0 = coordinates[0]
    x1, y1 = coordinates[1]
    for x, y in coordinates[2:]:
        # cross product of (p1 - p0) and (p - p0); zero means collinear
        if (x1 - x0) * (y - y0) != (y1 - y0) * (x - x0):
            return False
    return True

    for i, ch in enumerate(s):
        run = run + 1 if i > 0 and ch == s[i - 1] else 1   # length of current run
        total = (total + run) % MOD    # run substrings end here; reduce now

class Solution:
    def __init__(self, head: ListNode | None):
        self.head = head

    def get_random(self) -> int:
        chosen, i, node = 0, 0, self.head
        while node:
            i += 1
            if random.randrange(i) == 0:   # true with probability 1/i
                chosen = node.val          # replace the kept value
            node = node.next
        return chosen
```

class Solution:
    def __init__(self, nums: list[int]):
        self.orig = nums[:]

    def reset(self) -> list[int]:
        return self.orig[:]

    def shuffle(self) -> list[int]:

        for i in range(n):
            j = random.randint(i, n - 1)       # j in [i, n-1], both included
            arr[i], arr[j] = arr[j], arr[i]

@@ bitarith

def get_sum(a: int, b: int) -> int:
    mask = 0xFFFFFFFF                  # keep only 32 bits
    while b != 0:
        carry = ((a & b) << 1) & mask  # where both bits are 1, carry to the left
        a = (a ^ b) & mask             # sum without carries
        b = carry
    # if bit 31 is set, the number is negative in 32-bit two's complement
    return a if a <= 0x7FFFFFFF else ~(a ^ mask)
```

        m = 0
        for ch in w:
            m |= 1 << (ord(ch) - ord("a"))   # set the bit for this letter

def game_of_life(board: list[list[int]]) -> None:
    m, n = len(board), len(board[0])
    for r in range(m):
        for c in range(n):
            live = 0
            for dr in (-1, 0, 1):
                for dc in (-1, 0, 1):
                    if (dr or dc) and 0 <= r + dr < m and 0 <= c + dc < n:
                        live += board[r + dr][c + dc] & 1    # old state only
            if board[r][c] & 1:
                if live in (2, 3):
                    board[r][c] |= 2                         # stays alive
            elif live == 3:
                board[r][c] |= 2                             # is born
    for r in range(m):
        for c in range(n):
            board[r][c] >>= 1                                # new state becomes bit 0
```
